        var last2 = parts.slice(-2).join('.');
        if (twoLevel.indexOf(last2) !== -1 && parts.length >= 3) return parts.slice(-3).join('.');
        return parts.slice(-2).join('.');
    }

        if (st === STATE.GOOGLE_CLICK && age > CFG.watchdogMs) { setState(STATE.SCAN_BTN); return; }
        if (st === STATE.SCAN_BTN && age > 45000) { S.set('stateSetAt', Date.now().toString()); }
        if (st === STATE.WAIT_COUNTDOWN && age > 300000) { setState(STATE.SCAN_BTN); return; }

            if (S.get('backDone') !== '1') { S.set('backDone', '1'); var gt = S.get('gtrafficUrl'); if (gt) safeNavigate(gt); }
        }

    function onDomainListClick(e) {
        var target = e.target;
        if (!target || !target.closest) return;
        var row = target.closest('.mp-domain-item');
        if (!row) return;
        var d = row.getAttribute('data-domain');
        if (!d) return;
        var action = target.getAttribute && target.getAttribute('data-action');
        if (action === 'use') {
            e.stopPropagation();
            if (UI.domainInput) UI.domainInput.value = d;
            return;
        }
        if (action === 'del') {
            e.stopPropagation();
            DOMAINS.remove(d);
            renderDomainList();
            return;
        }
        if (UI.domainInput) UI.domainInput.value = d;
    }

    function copyToClipboard(text) {
        if (!text) return false;
        var tried = false;
        try {
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(text).then(function() {
                    log('COPY OK');
                    S.set('lastCopyOk', '1');
                    S.set('lastCopyText', text);
                }).catch(function(e) {
                    if (copyViaExecCommand(text)) { S.set('lastCopyOk', '1'); S.set('lastCopyText', text); }
                    else { S.set('lastCopyOk', '0'); S.set('lastCopyText', text); }
                });
                tried = true;
            }
        } catch (e) {}
        if (!tried) {
            if (copyViaExecCommand(text)) { S.set('lastCopyOk', '1'); S.set('lastCopyText', text); }
            else { S.set('lastCopyOk', '0'); S.set('lastCopyText', text); }
        }
        return true;
    }

    function copyViaExecCommand(text) {
        try {
            var ta = document.createElement('textarea');
            ta.value = text;
            ta.setAttribute('readonly', '');
            ta.style.cssText = 'position:fixed;top:0;left:0;width:2px;height:2px;padding:0;border:none;outline:none;box-shadow:none;background:transparent;opacity:0.01;z-index:-1;';
            document.body.appendChild(ta);
            ta.focus();
            ta.select();
            try { ta.setSelectionRange(0, text.length); } catch (e) {}
            var ok = false;
            try { ok = document.execCommand('copy'); } catch (e) {}
            document.body.removeChild(ta);
            if (ok) { log('COPY OK (exec)'); return true; }
            return false;
        } catch (e) { return false; }
    }

    function isBlueish(c) {
        if (!c) return false;
        if (c.b < 110) return false;
        if (c.r > c.b) return false;
        if (c.g > c.b + 25) return false;
        if (c.b - c.r < 30) return false;
        return true;
    }

    function getBgRGBA(el) {
        var c = tryGetBg(el);
        if (c && c.a > 0.5) return c;
        var p = el.parentElement;
        var depth = 0;
        while (p && depth < 5) {
            var pc = tryGetBg(p);
            if (pc && pc.a > 0.5) return pc;
            p = p.parentElement;
            depth++;
        }
        return null;
    }

    function tryGetBg(el) {
        if (!el) return null;
        try {
            var st = window.getComputedStyle(el);
            var bg = st.backgroundColor || '';
            var m = bg.match(/rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)(?:\s*,\s*([\d.]+))?/);
            if (m) {
                var a = m[4] !== undefined ? parseFloat(m[4]) : 1;
                return { r: parseInt(m[1],10), g: parseInt(m[2],10), b: parseInt(m[3],10), a: a };
            }
            var bgImg = st.backgroundImage || '';
            var m2 = bgImg.match(/rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)(?:\s*,\s*([\d.]+))?/);
            if (m2) {
                var a2 = m2[4] !== undefined ? parseFloat(m2[4]) : 1;
                return { r: parseInt(m2[1],10), g: parseInt(m2[2],10), b: parseInt(m2[3],10), a: a2 };
            }
        } catch (e) {}
        return null;
    }

    function isRound(el) {
        var w = el.offsetWidth, h = el.offsetHeight;
        if (w < 30 || w > 200) return false;
        if (h < 30 || h > 200) return false;
        if (Math.abs(w - h) > 15) return false;
        return true;
    }
    function isExcluded(el) {
        var cls = (el.className || '').toString().toLowerCase();
        var id = (el.id || '').toString().toLowerCase();
        var c = ' ' + cls + ' ' + id + ' ';
        var ban = ['logo','brand','avatar','banner','header','nav','menu','footer','navbar','topbar','toolbar','copyright'];
        for (var i = 0; i < ban.length; i++) {
            if (c.indexOf(' ' + ban[i] + ' ') !== -1) return true;
            if (c.indexOf(' ' + ban[i] + '-') !== -1) return true;
            if (c.indexOf('-' + ban[i] + ' ') !== -1) return true;
        }
        return false;
    }

    function findByPixelScan() {
        if (!document.body) return null;
        var vw = window.innerWidth, vh = window.innerHeight;
        var step = 30;
        var seen = {};
        var candidates = [];
        for (var y = 40; y < vh - 40; y += step) {
            for (var x = 40; x < vw - 40; x += step) {
                try {
                    var el = document.elementFromPoint(x, y);
                    if (!el) continue;
                    if (el.closest && el.closest('#mp-root')) continue;
                    if (el.closest && el.closest('#mp-fab')) continue;
                    if (isExcluded(el)) continue;
                    if (!isRound(el)) {
                        var p = el.parentElement; var cnt = 0;
                        while (p && cnt < 3) {
                            if (isExcluded(p)) break;
                            if (isRound(p)) { el = p; break; }
                            p = p.parentElement; cnt++;
                        }
                        if (!isRound(el)) continue;
                    }
                    var key = el.tagName + '#' + (el.id||'') + '.' + (el.className||'').toString().slice(0,20);
                    if (seen[key]) continue;
                    seen[key] = true;
                    var score = 0;
                    var bg = getBgRGBA(el);
                    if (isBlueish(bg)) score += 100;
                    try { if (el.querySelector('svg, img, canvas, text')) score += 60; } catch (e) {}
                    if (el.tagName === 'SVG' || el.tagName === 'IMG' || el.tagName === 'CANVAS') score += 50;
                    var txt = (el.textContent || '').trim();
                    if (txt === 'g' || txt === 'G') score += 200;
                    else if (/^\d{1,4}$/.test(txt)) score += 200;
                    else if (/^g\d{1,4}$/i.test(txt)) score += 200;
                    var r = el.getBoundingClientRect();
                    var cx = r.left + r.width / 2; var cy = r.top + r.height / 2;
                    var d = Math.abs(cx - vw/2)/vw + Math.abs(cy - vh/2)/vh;
                    score += Math.max(0, 40 - d * 40);
                    if (score >= 100) candidates.push({ el: el, score: score });
                } catch (e) {}
            }
        }
        if (!candidates.length) return null;
        candidates.sort(function(a,b){ return b.score - a.score; });
        return candidates[0].el;
    }

    function findByHeuristic() {
        try {
            var byId = document.querySelectorAll('#avt-btn, [id*="avt" i], [class*="avt" i]');
            for (var i = 0; i < byId.length; i++) {
                var e = byId[i];
                if (e.closest && e.closest('#mp-root')) continue;
                var w = e.offsetWidth || e.clientWidth;
                var h = e.offsetHeight || e.clientHeight;
                if (w >= 30 && w <= 250 && Math.abs(w-h) <= 20) return e;
            }
        } catch (e) {}
        try {
            var svgs = document.querySelectorAll('svg');
            for (var j = 0; j < svgs.length; j++) {
                var svg = svgs[j];
                if (svg.closest && svg.closest('#mp-root')) continue;
                var sw = svg.offsetWidth || svg.clientWidth;
                var sh = svg.offsetHeight || svg.clientHeight;
                if (sw < 30 || sw > 250) continue;
                if (Math.abs(sw-sh) > 20) continue;
                if (svg.offsetParent === null) continue;
                try {
                    var txtEl = svg.querySelector('text, tspan');
                    if (txtEl) {
                        var tt = (txtEl.textContent || '').trim();
                        if (/^g?\d+$/i.test(tt) || /^[A-Za-z0-9_\-]{4,24}$/.test(tt)) return svg;
                    }
                } catch (e) {}
                try {
                    var circles = svg.querySelectorAll('circle');
                    if (circles.length > 0 && sw >= 40) return svg;
                } catch (e) {}
            }
        } catch (e) {}
        try {
            var all = document.querySelectorAll('div, span, button, a, i, b, strong');
            for (var k = 0; k < all.length; k++) {
                var e2 = all[k];
                if (e2.closest && e2.closest('#mp-root')) continue;
                if (e2.closest && e2.closest('#mp-fab')) continue;
                var w2 = e2.offsetWidth || 0;
                var h2 = e2.offsetHeight || 0;
                if (w2 < 30 || w2 > 200) continue;
                if (Math.abs(w2-h2) > 15) continue;
                var txt2 = (e2.textContent || '').trim();
                if (txt2 !== 'g' && txt2 !== 'G') continue;
                return e2;
            }
        } catch (e) {}
        try {
            var all2 = document.querySelectorAll('div, span, button, a, i');
            for (var m = 0; m < all2.length; m++) {
                var e3 = all2[m];
                if (e3.closest && e3.closest('#mp-root')) continue;
                if (e3.closest && e3.closest('#mp-fab')) continue;
                var w3 = e3.offsetWidth || 0;
                var h3 = e3.offsetHeight || 0;
                if (w3 < 40 || w3 > 150) continue;
                if (Math.abs(w3-h3) > 15) continue;
                var bg3 = getBgRGBA(e3);
                if (!isBlueish(bg3)) continue;
                return e3;
            }
        } catch (e) {}
        return null;
    }

        /cho\s*(\d+)\s*s\s*\(\s*(\d+)\s*\/\s*(\d+)\s*\)/i,
        /cho\s*(\d+)\s*giay/i, /cho\s*(\d+)\s*s\b/i,
        /con[:\s]*(\d+)/i, /wait[:\s]*(\d+)/i, /time[:\s]*(\d+)/i,
        /(\d+)\s*seconds?/i, /(\d+)\s*giay/i
    ];
    function parseCd(t) {
        if (!t) return null;
        for (var p = 0; p < CD_PATTERNS.length; p++) {
            var m = t.match(CD_PATTERNS[p]);
            if (m) {
                var s = parseInt(m[1],10);
                if (s >= 0 && s <= 999) return { sec: s, phase: m[2]?parseInt(m[2],10):1, total: m[3]?parseInt(m[3],10):2 };
            }
        }
        return null;
    }

    function readCountdown() {
        var btn = cachedBtn;
        if (!btn || !document.contains(btn)) btn = findGreenGButton();

        if (btn) {
            var n = readBtnNumber(btn);
            if (n !== null) {
                if (S.get('cdStartSec') === '0' || !S.get('cdStartSec')) {
                    S.set('cdStartSec', n.toString());
                    S.set('cdStartAt', Date.now().toString());
                }
                if (n === 0) S.set('countdownRealZero', '1');
                else S.set('countdownRealZero', '0');
                return { sec: n, phase: 1, total: 2, src: 'btn' };
            }
        }

        try {
            var names = ['countdown','timer','timeLeft','timeleft','remaining','remain','cd','counter','count','sec','seconds'];
            for (var i = 0; i < names.length; i++) {
                var v = window[names[i]];
                if (typeof v === 'number' && v >= 0 && v <= 999) {
                    if (v === 0) S.set('countdownRealZero', '1');
                    else S.set('countdownRealZero', '0');
                    return { sec: v, phase: 1, total: 2, src: 'win' };
                }
                if (typeof v === 'string' && /^\d{1,4}$/.test(v)) {
                    var nv = parseInt(v, 10);
                    if (nv === 0) S.set('countdownRealZero', '1');
                    else S.set('countdownRealZero', '0');
                    return { sec: nv, phase: 1, total: 2, src: 'win' };
                }
            }
        } catch (e) {}

        try {
            if (btn) {
                var els = btn.querySelectorAll('text, tspan');
                for (var j = 0; j < els.length; j++) {
                    var t = (els[j].textContent || '').replace(/\s+/g, '').trim();
                    var mm = t.match(/^g?(\d{1,4})$/i);
                    if (mm) {
                        var n2 = parseInt(mm[1], 10);
                        if (n2 >= 0 && n2 <= 999) {
                            if (n2 === 0) S.set('countdownRealZero', '1');
                            else S.set('countdownRealZero', '0');
                            return { sec: n2, phase: 1, total: 2, src: 'svgtext' };
                        }
                    }
                }
            }
        } catch (e) {}

        try {
            var bt = (document.body && document.body.innerText) || '';
            var r = parseCd(bt);
            if (r) {
                if (r.sec === 0) S.set('countdownRealZero', '1');
                else S.set('countdownRealZero', '0');
                return { sec: r.sec, phase: r.phase, total: r.total, src: 'page' };
            }
        } catch (e) {}

        var startSec = parseInt(S.get('cdStartSec', '0'), 10);
        var startAt = parseInt(S.get('cdStartAt', '0'), 10);
        if (startSec > 0 && startAt > 0) {
            var elapsed = Math.floor((Date.now() - startAt) / 1000);
            var remain = startSec - elapsed;
            if (remain < 0) remain = 0;
            return { sec: remain, phase: 1, total: 2, src: 'self' };
        }
        var clickAt = parseInt(S.get('btnClickAt', '0'), 10);
        if (clickAt > 0) {
            var elapsed2 = Math.floor((Date.now() - clickAt) / 1000);
            var remain2 = CFG.defaultCountdown - elapsed2;
            if (remain2 < 0) remain2 = 0;
            return { sec: remain2, phase: 1, total: 2, src: 'click' };
        }
        return null;
    }

    function isValidCodeShape(s) {
        if (!s) return false;
        if (s.length !== 8) return false;
        if (!/^[A-Za-z0-9]+$/.test(s)) return false;
        if (!/[0-9]/.test(s)) return false;
        if (!/[A-Za-z]/.test(s)) return false;

        var digits = (s.match(/[0-9]/g) || []).length;
        if (digits < 1 || digits > 7) return false;
        var letters = (s.match(/[A-Za-z]/g) || []).length;
        if (letters < 1 || letters > 7) return false;
        if (/^(.)\1{7}$/.test(s)) return false;
        if (/^(01234567|12345678|abcdefgh|ABCDEFGH|00000000|11111111)$/i.test(s)) return false;
        return true;
    }

    function scoreCodeCandidate(el, code) {
        var score = 0;
        try {
            if (cachedBtn && document.contains(cachedBtn)) {
                try {
                    var r1 = el.getBoundingClientRect();
                    var r2 = cachedBtn.getBoundingClientRect();
                    var dx = (r1.left + r1.width/2) - (r2.left + r2.width/2);
                    var dy = (r1.top + r1.height/2) - (r2.top + r2.height/2);
                    var dist = Math.sqrt(dx*dx + dy*dy);
                    if (dist < 150) score += 200;
                    else if (dist < 300) score += 150;
                    else if (dist < 500) score += 100;
                    else if (dist < 800) score += 50;
                } catch (e) {}
            }
            try {
                var bg = getBgRGBA(el);
                if (isBlueish(bg)) score += 500;
            } catch (e) {}
            try {
                var st = window.getComputedStyle(el);
                var fs = parseFloat(st.fontSize) || 0;
                if (fs >= 20) score += 60;
                else if (fs >= 16) score += 40;
                else if (fs >= 14) score += 20;
                if (st.fontWeight === 'bold' || parseInt(st.fontWeight,10) >= 600) score += 30;
            } catch (e) {}
            try {
                var cls = ((el.className || '') + ' ' + (el.id || '')).toLowerCase();
                if (cls.indexOf('code') !== -1 || cls.indexOf('result') !== -1 || cls.indexOf('ma-') !== -1 || cls.indexOf('-ma') !== -1) score += 80;
            } catch (e) {}
            try {
                var par = el.closest && el.closest('button, a, [role="button"]');
                if (par) score += 40;
            } catch (e) {}
            try {
                var w = el.offsetWidth || 0;
                var h = el.offsetHeight || 0;
                if (w >= 60 && h >= 30) score += 30;
                if (w >= 100 && h >= 40) score += 20;
            } catch (e) {}
        } catch (e) {}
        return score;
    }

    function scanPageForCode() {
        if (getState() !== STATE.GET_CODE) {
            return null;
        }

        var candidates = [];
        var seen = {};

        try {
            var all = document.querySelectorAll('div, span, button, a, p, strong, h1, h2, h3, h4, td, th, li, label, section, article');
            for (var i = 0; i < all.length; i++) {
                var el = all[i];
                if (el.closest && el.closest('#mp-root')) continue;
                if (el.closest && el.closest('#mp-fab')) continue;
                if (cachedBtn && el === cachedBtn) continue;
                if (el.closest && el.closest('#avt-btn, [id*="avt" i]')) continue;

                var txt = (el.textContent || '').replace(/\s+/g, '').trim();
                if (!txt || txt.length !== 8) continue;
                if (!/^[A-Za-z0-9]+$/.test(txt)) continue;
                if (isBlacklistedCode(txt)) continue;
                if (!isValidCodeShape(txt)) continue;

                var key = txt + '@P1@' + i;
                if (seen[key]) continue;
                seen[key] = true;

                var bg = getBgRGBA(el);
                if (isBlueish(bg)) {
                    candidates.push({ code: txt, el: el, score: 500 });
                    log('✓ CODE NỀN XANH: ' + txt);
                    continue;
                }

                var score = scoreCodeCandidate(el, txt);
                if (score >= CFG.minCodeScore) {
                    candidates.push({ code: txt, el: el, score: score });
                }
            }
        } catch (e) {}

        try {
            var walker = document.createTreeWalker(document.body || document, NodeFilter.SHOW_TEXT, null, false);
            var node; var count = 0;
            while ((node = walker.nextNode())) {
                count++;
                if (count > 50000) break;
                var t = (node.textContent || '').replace(/\s+/g, '').trim();
                if (!t || t.length !== 8) continue;
                if (!/^[A-Za-z0-9]+$/.test(t)) continue;
                if (isBlacklistedCode(t)) continue;
                if (!isValidCodeShape(t)) continue;

                var par = node.parentElement;
                if (!par) continue;
                if (par.closest && par.closest('#mp-root')) continue;
                if (cachedBtn && par === cachedBtn) continue;
                if (par.closest && par.closest('#avt-btn, [id*="avt" i]')) continue;

                var key2 = t + '@P2@' + count;
                if (seen[key2]) continue;
                seen[key2] = true;

                var bg2 = getBgRGBA(par);
                if (isBlueish(bg2)) {
                    candidates.push({ code: t, el: par, score: 500 });
                    log('✓ CODE NỀN XANH (parent): ' + t);
                    continue;
                }

                var score2 = scoreCodeCandidate(par, t);
                if (score2 >= CFG.minCodeScore) {
                    candidates.push({ code: t, el: par, score: score2 });
                }
            }
        } catch (e) {}

        if (!candidates.length) {
            log('SCAN: khong tim thay ung vien nao');
            return null;
        }

        candidates.sort(function(a, b) { return b.score - a.score; });

        log('SCAN: ' + candidates.length + ' ung vien. Top: ' + candidates[0].code + ' score=' + candidates[0].score);
        return candidates[0].code;
    }

    function startGoogleObserver() {
        if (!IS_TOP || !isGoogle) return;
        if (googleObserver) return;
        try {
            googleObserver = new MutationObserver(function() {
                if (stopRequested) return;
                var st = getState();
                if (st !== STATE.GOOGLE_SEARCH && st !== STATE.GOOGLE_CLICK) return;
                var link = findGoogleResult();
                if (link) {
                    var href = link.href || '';
                    log('✓ MUTATION: click ngay → ' + href);
                    setState(STATE.GOOGLE_CLICK);
                    S.set('targetUrl', href);
                    try { link.click(); } catch (e) {}
                    try { link.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true })); } catch (e) {}
                }
            });
            googleObserver.observe(document.documentElement || document.body, { childList: true, subtree: true });
        } catch (e) {}
    }

    function handleGoogle() {
        if (!IS_TOP) return Promise.resolve();
        var st = getState();
        if (st !== STATE.GOOGLE_SEARCH && st !== STATE.GOOGLE_CLICK) return Promise.resolve();
        var searchTerm = S.get('keyword') || S.get('domain');
        if (!searchTerm) return Promise.resolve();
        var tries = parseInt(S.get('googleClickTries', '0'), 10);
        tries++;
        S.set('googleClickTries', tries.toString());
        var link = findGoogleResult();
        if (link) {
            var href = link.href || '';
            setState(STATE.GOOGLE_CLICK);
            S.set('targetUrl', href);
            log('✓ CLICK TRANG DAU TIEN: ' + href);
            try { link.click(); } catch (e) {}
            try { link.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true })); } catch (e) {}
            try {
                var p = link.parentElement;
                if (p && p.tagName === 'A') p.click();
                var gp = link.closest('a');
                if (gp && gp !== link) gp.click();
            } catch (e) {}
            return Promise.resolve();
        }
        if (tries >= 15) {
            var cleanDomain = (S.get('domain', '')||'').toLowerCase().replace(/^https?:\/\//,'').replace(/^www\./,'').replace(/\/.*$/,'').trim();
            if (cleanDomain && cleanDomain.indexOf('.') !== -1) {
                var url = 'https://' + cleanDomain;
                log('handleGoogle: fallback navigate → ' + url);
                S.set('targetUrl', url);
                setState(STATE.SCAN_BTN);
                safeNavigate(url);
            }
            return Promise.resolve();
        }
        return Promise.resolve();
    }

            } catch (e) {}
            if (sc > bestScore) { bestScore = sc; best = el; }
        }
        return best;
    }

    function forceFill(inp, code) {
        try {
            inp.focus();
            try { inp.click(); } catch (e) {}
            try {
                var d = Object.getOwnPropertyDescriptor(inp, 'value');
                if (d && d.set) d.set.call(inp, code);
                else {
                    var dP = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value');
                    if (dP && dP.set) dP.set.call(inp, code);
                    else inp.value = code;
                }
            } catch (e) { try { inp.value = code; } catch (e2) {} }
            try { inp.dispatchEvent(new Event('input', { bubbles: true, cancelable: true })); } catch (e) {}
            try { inp.dispatchEvent(new Event('change', { bubbles: true, cancelable: true })); } catch (e) {}
            return inp.value === code;
        } catch (e) { return false; }
    }

    function charByCharFill(inp, code) {
        try {
            inp.focus();
            try { inp.click(); } catch (e) {}
            try {
                var dP = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value');
                if (dP && dP.set) dP.set.call(inp, '');
                else inp.value = '';
            } catch (e) { inp.value = ''; }
            try { inp.dispatchEvent(new Event('input', { bubbles: true })); } catch (e) {}

            for (var i = 0; i < code.length; i++) {
                var ch = code[i];
                try { inp.dispatchEvent(new KeyboardEvent('keydown', { bubbles: true, cancelable: true, key: ch, keyCode: ch.charCodeAt(0) })); } catch (e) {}
                try { inp.dispatchEvent(new KeyboardEvent('keypress', { bubbles: true, cancelable: true, key: ch, charCode: ch.charCodeAt(0) })); } catch (e) {}
                try {
                    var d = Object.getOwnPropertyDescriptor(inp, 'value');
                    if (d && d.set) d.set.call(inp, inp.value + ch);
                    else {
                        var dP2 = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value');
                        if (dP2 && dP2.set) dP2.set.call(inp, inp.value + ch);
                        else inp.value += ch;
                    }
                } catch (e) { inp.value += ch; }
                try { inp.dispatchEvent(new Event('input', { bubbles: true })); } catch (e) {}
                try { inp.dispatchEvent(new KeyboardEvent('keyup', { bubbles: true, cancelable: true, key: ch, keyCode: ch.charCodeAt(0) })); } catch (e) {}
            }
            try { inp.dispatchEvent(new Event('change', { bubbles: true })); } catch (e) {}
            try { inp.dispatchEvent(new CompositionEvent('compositionend', { bubbles: true, data: code })); } catch (e) {}
            return inp.value === code;
        } catch (e) { return false; }
    }

    function pasteEventFill(inp, code) {
        try {
            inp.focus();
            try { inp.select(); } catch (e) {}
            try {
                var dt = new DataTransfer();
                dt.setData('text/plain', code);
                var pe = new ClipboardEvent('paste', { bubbles: true, cancelable: true, clipboardData: dt });
                inp.dispatchEvent(pe);
            } catch (e) {}
            try {
                var d = Object.getOwnPropertyDescriptor(inp, 'value');
                if (d && d.set) d.set.call(inp, code);
                else inp.value = code;
            } catch (e) { inp.value = code; }
            try { inp.dispatchEvent(new InputEvent('input', { bubbles: true, data: code, inputType: 'insertFromPaste' })); } catch (e) {}
            try { inp.dispatchEvent(new Event('change', { bubbles: true, cancelable: true })); } catch (e) {}
            return inp.value === code;
        } catch (e) { return false; }
    }

    async function nativeTypingFill(inp, code) {
        try {
            inp.focus();
            try { inp.click(); } catch (e) {}
            try { inp.select(); } catch (e) {}

            try {
                inp.setSelectionRange(0, inp.value.length);
                document.execCommand('delete', false, null);
            } catch (e) {
                try {
                    var dP = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value');
                    if (dP && dP.set) dP.set.call(inp, '');
                    else inp.value = '';
                } catch (e2) { inp.value = ''; }
            }
            try { inp.dispatchEvent(new Event('input', { bubbles: true })); } catch (e) {}

            var inserted = false;
            try { inserted = document.execCommand('insertText', false, code); } catch (e) { inserted = false; }
            if (inserted && inp.value === code) {
                try { inp.dispatchEvent(new Event('change', { bubbles: true })); } catch (e) {}
                return true;
            }

            try {
                var dP2 = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value');
                if (dP2 && dP2.set) dP2.set.call(inp, '');
                else inp.value = '';
            } catch (e) { inp.value = ''; }
            try { inp.dispatchEvent(new Event('input', { bubbles: true })); } catch (e) {}

            for (var i = 0; i < code.length; i++) {
                var ch = code[i];
                try { inp.dispatchEvent(new KeyboardEvent('keydown', { bubbles: true, cancelable: true, key: ch, code: 'Key' + ch.toUpperCase(), keyCode: ch.charCodeAt(0), which: ch.charCodeAt(0) })); } catch (e) {}
                try { inp.dispatchEvent(new InputEvent('beforeinput', { bubbles: true, cancelable: true, inputType: 'insertText', data: ch })); } catch (e) {}
                var ok = false;
                try { ok = document.execCommand('insertText', false, ch); } catch (e) { ok = false; }
                if (!ok) {
                    try {
                        var d = Object.getOwnPropertyDescriptor(inp, 'value');
                        var nv = (inp.value || '') + ch;
                        if (d && d.set) d.set.call(inp, nv);
                        else {
                            var dP3 = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value');
                            if (dP3 && dP3.set) dP3.set.call(inp, nv);
                            else inp.value = nv;
                        }
                    } catch (e) { inp.value = (inp.value || '') + ch; }
                }
                try { inp.dispatchEvent(new InputEvent('input', { bubbles: true, cancelable: true, inputType: 'insertText', data: ch })); } catch (e) {}
                try { inp.dispatchEvent(new KeyboardEvent('keyup', { bubbles: true, cancelable: true, key: ch, code: 'Key' + ch.toUpperCase(), keyCode: ch.charCodeAt(0), which: ch.charCodeAt(0) })); } catch (e) {}
                await sleep(rand(CFG.nativePasteDelayMin, CFG.nativePasteDelayMax));
            }

            try { inp.dispatchEvent(new Event('change', { bubbles: true })); } catch (e) {}
            try { inp.dispatchEvent(new CompositionEvent('compositionend', { bubbles: true, data: code })); } catch (e) {}
            return inp.value === code;
        } catch (e) { return false; }
    }

    function fillLoop(code, callback) {
        var attempts = 0;

        function nextAttempt() {
            if (stopRequested || fillDone || S.get('confirmed') === '1' || S.get('hardStop') === '1') {
                if (callback) callback(false);
                return;
            }

            var inp = findCodeInputCached();
            if (!inp) {
                attempts++;
                if (attempts % 20 === 0) log('fillLoop: chua tim thay input (lan ' + attempts + ')');
                if (attempts < CFG.fillLoopMax) setTimeout(nextAttempt, CFG.fillLoopDelay);
                else { if (callback) callback(false); }
                return;
            }

            attempts++;

            var tier = (attempts % 4);
            if (tier === 1) {
                forceFill(inp, code);
            } else if (tier === 2) {
                charByCharFill(inp, code);
            } else if (tier === 3) {
                pasteEventFill(inp, code);
            } else {
                nativeTypingFill(inp, code).then(function() {
                    verifyHold(code, callback, attempts);
                });
                return;
            }

            setTimeout(function() {
                verifyHold(code, callback, attempts);
            }, 80);
        }

        function verifyHold(code, callback, attemptNum) {
            if (stopRequested || fillDone || S.get('confirmed') === '1' || S.get('hardStop') === '1') {
                if (callback) callback(false);
                return;
            }
            var startAt = Date.now();
            var lastValue = '';
            var stableCount = 0;

            function checkTick() {
                if (stopRequested || fillDone || S.get('confirmed') === '1' || S.get('hardStop') === '1') {
                    if (callback) callback(false); return;
                }
                var inp = findCodeInputCached();
                if (!inp) {
                    if (attemptNum < CFG.fillLoopMax) setTimeout(nextAttempt, CFG.fillLoopDelay);
                    else callback(false);
                    return;
                }
                var cur = (inp.value || '').trim();

                if (cur === code) {
                    stableCount++;
                    lastValue = cur;
                    if (Date.now() - startAt >= CFG.fillHoldVerifyMs) {
                        if (callback) callback(true);
                        return;
                    }
                    setTimeout(checkTick, CFG.fillVerifyIntervalMs);
                } else {
                    if (attemptNum < CFG.fillLoopMax) {
                        setTimeout(nextAttempt, 100);
                    } else {
                        if (lastValue === code) {
                            callback(true);
                        } else {
                            callback(false);
                        }
                    }
                }
            }
            checkTick();
        }

        nextAttempt();
    }

    function startWaitingForGetLink() {
        S.set('waitingGetLink', '1');
        getLinkStartAt = Date.now();
        S.set('getLinkStartAt', getLinkStartAt.toString());
        getLinkClickDone = false;

        var minWaitMs = CFG.minWaitBeforeSearchMs;
        var searchStartAt = Date.now() + minWaitMs;
        log('★ BẮT ĐẦU QUÉT NÚT LẤY LINK (bắt đầu sau ' + minWaitMs + 'ms, timeout ' + (CFG.getLinkTimeoutMs/1000) + 's)');
        if (IS_TOP) showGuide('<b>★ ĐÃ NỘP MÃ ★</b><br>Đang chờ nút LẤY LINK xuất hiện...');

        if (getLinkPollTimer) { try { clearInterval(getLinkPollTimer); } catch (e) {} getLinkPollTimer = null; }
        if (getLinkObserver) { try { getLinkObserver.disconnect(); } catch (e) {} getLinkObserver = null; }

        function tryClickIfReady(source) {
            if (getLinkClickDone) return false;
            if (Date.now() < searchStartAt) return false;
            var btn = findGetLinkButton();
            if (!btn) return false;
            getLinkClickDone = true;
            var elapsed = Date.now() - getLinkStartAt;
            log('★ ' + source + ': ĐÃ CLICK NÚT LẤY LINK (sau ' + elapsed + 'ms) - STOP NGAY');
            if (IS_TOP) showGuide('<b>★ ĐÃ CLICK NÚT LẤY LINK ★</b><br>Hoàn tất!');
            clickGetLinkButton(btn);
            S.set('getLinkClicked', '1');
            if (getLinkObserver) { try { getLinkObserver.disconnect(); } catch (e) {} getLinkObserver = null; }
            if (getLinkPollTimer) { try { clearInterval(getLinkPollTimer); } catch (e) {} getLinkPollTimer = null; }
            S.del('waitingGetLink');
            stopAll();
            return true;
        }

        // Lớp 1: MutationObserver - bắt NGAY khi DOM đổi
        try {
            getLinkObserver = new MutationObserver(function() {
                tryClickIfReady('MUTATION');
            });
            getLinkObserver.observe(document.body || document.documentElement, { childList: true, subtree: true });
        } catch (e) {}

        // Lớp 2: Poll cực nhanh - 5ms
        getLinkPollTimer = setInterval(function() {
            if (getLinkClickDone) {
                clearInterval(getLinkPollTimer); getLinkPollTimer = null;
                return;
            }
            var now = Date.now();
            if (now - getLinkStartAt > CFG.getLinkTimeoutMs) {
                log('⚠ Hết ' + (CFG.getLinkTimeoutMs/1000) + 's không thấy nút LẤY LINK');
                getLinkClickDone = true;
                clearInterval(getLinkPollTimer); getLinkPollTimer = null;
                if (getLinkObserver) { try { getLinkObserver.disconnect(); } catch (e) {} getLinkObserver = null; }
                S.del('waitingGetLink');
                stopAll();
                return;
            }
            tryClickIfReady('POLL');
        }, CFG.getLinkPollMs);
    }

    function fillAndConfirm(force) {
        if (!IS_TOP || !isGtraffic) return false;
        if (S.get('hardStop') === '1') { return false; }
        if (fillDone && !force) return true;
        if (fillRunning) return true;
        fillRunning = true;

        var code = getCodeFromPanel();
        if (!code || !isValidCodeShape(code) || isBlacklistedCode(code)) {
            var sc = getCodeFromStorage();
            if (sc && isValidCodeShape(sc) && !isBlacklistedCode(sc)) {
                code = sc;
            }
        }

        if (!code || !isValidCodeShape(code) || isBlacklistedCode(code)) {
            fillRetries++;
            fillRunning = false;
            if (fillRetries < CFG.fillRetryMax) {
                setTimeout(function() { fillAndConfirm(force); }, CFG.fillRetryDelay);
            }
            return false;
        }

        S.set('filled', '1');
        S.set('code', code);

        if (!codeShown && UI) {
            codeShown = true;
            UI.code.classList.add('show');
            UI.codeVal.textContent = code;
            showGuide('<b>★ ĐANG DÁN MÃ ★</b><br>Mã: <b>' + code + '</b>');
        }

        fillLoop(code, function(success) {
            if (stopRequested) { fillRunning = false; return; }
            if (S.get('hardStop') === '1') { fillRunning = false; return; }

            if (success) {
                showGuide('<b>★ ĐÃ DÁN MÃ ★</b><br>Đang nộp...');
                setTimeout(function() {
                    if (stopRequested) { fillRunning = false; return; }
                    if (S.get('hardStop') === '1') { fillRunning = false; return; }

                    clickConfirmNow();

                    var checkCount = 0;
                    var maxChecks = CFG.postSubmitCheckMax;

                    function checkLoop() {
                        if (S.get('hardStop') === '1') { fillRunning = false; return; }
                        checkCount++;
                        var result = checkSubmitResult();

                        if (result === 'success') {
                            finishSuccess('success');
                            return;
                        }

                        if (result === 'error') {
                            S.set('confirmed', '0');
                            fillDone = false;
                            fillRunning = false;
                            var inp = findCodeInputCached();
                            if (inp) {
                                try {
                                    var dP = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value');
                                    if (dP && dP.set) dP.set.call(inp, '');
                                    else inp.value = '';
                                } catch (e) { inp.value = ''; }
                                try { inp.dispatchEvent(new Event('input', { bubbles: true })); } catch (e) {}
                            }
                            setTimeout(function() { fillAndConfirm(true); }, 800);
                            return;
                        }

                        if (result === 'pending') {
                            if (checkCount < maxChecks) {
                                setTimeout(checkLoop, CFG.postSubmitCheckMs);
                            } else {
                                finishSuccess('pending-timeout');
                            }
                            return;
                        }

                        if (checkCount < maxChecks) {
                            setTimeout(checkLoop, CFG.postSubmitCheckMs);
                        } else {
                            finishSuccess('unknown-timeout');
                        }
                    }
                    setTimeout(checkLoop, CFG.postSubmitCheckMs);
                }, 100);
            } else {
                clickConfirmNow();
                S.set('confirmed', '1');
                S.set('hardStop', '1');
                fillDone = true;
                setState(STATE.DONE);
                fillRunning = false;
                if (gtrafficCheckTimer) { clearInterval(gtrafficCheckTimer); gtrafficCheckTimer = null; }
                startWaitingForGetLink();
            }
        });

        return true;
    }

            var btn = findGetLinkButton();
            if (btn) {
                log('✓ CLICK NÚT "LẤY LINK" (boot path): ' + describeEl(btn));
                S.set('getLinkClicked', '1');
                clickGetLinkButton(btn);
            }
        }
        return Promise.resolve();
    }

    function goBackToGtraffic() {
        if (S.get('backDone') === '1') { return; }
        var gt = S.get('gtrafficUrl');
        if (!gt) return;
        S.set('backDone', '1');
        S.set('navigatingBack', '1');
        safeNavigate(gt);
    }

    function handleTarget() {
        if (isGtraffic || isGoogle) return Promise.resolve();
        var st = getState();
        if ((st === STATE.GOOGLE_SEARCH || st === STATE.GOOGLE_CLICK) && isRealTargetPage()) {
            setState(STATE.SCAN_BTN); return Promise.resolve();
        }
        if (st === STATE.SCAN_BTN) {
            var b = findGreenGButton();
            if (b) {
                var lc = parseInt(S.get('lastClickTime','0'), 10);
                if (Date.now() - lc > 3500) {
                    S.set('lastClickTime', Date.now().toString());
                    S.set('btnClickAt', Date.now().toString());
                    clickBtn(b);
                    setTimeout(function() {
                        if (getState() === STATE.SCAN_BTN) setState(STATE.WAIT_COUNTDOWN);
                    }, 2000);
                }
            }
            return Promise.resolve();
        }
        if (st === STATE.WAIT_COUNTDOWN) {
            var cd = readCountdown();
            if (cd) {
                var tot = parseInt(S.get('serverTotalSec','0'), 10);
                if (cd.sec > tot) { tot = cd.sec; S.set('serverTotalSec', tot.toString()); }
                if (tot === 0) tot = cd.sec || CFG.defaultCountdown;
                if (IS_TOP) showTimer(cd.sec, tot, 'B5: CHỜ ' + cd.sec + 's');
                S.set('lastServerSec', cd.sec.toString());

                if (cd.sec === 0) {
                    log('B5: countdown thực sự = 0 (src=' + cd.src + '), chuyen get-code');
                    setState(STATE.GET_CODE);
                    return Promise.resolve();
                }
                return Promise.resolve();
            }
            var b2 = cachedBtn || findGreenGButton();
            if (b2) {
                var bt2 = (b2.textContent || '').replace(/\s+/g, '').trim();
                if (bt2.length === 8 && /^[A-Za-z0-9]+$/.test(bt2) && !isBlacklistedCode(bt2) && isValidCodeShape(bt2)) {
                    log('B5: nut co code thật: ' + bt2);
                    setState(STATE.GET_CODE);
                    return Promise.resolve();
                }
            }
            var clickAt = parseInt(S.get('btnClickAt', '0'), 10);
            if (clickAt > 0 && IS_TOP) {
                var elapsed = Math.floor((Date.now() - clickAt) / 1000);
                var remain = Math.max(0, CFG.defaultCountdown - elapsed);
                showTimer(remain, CFG.defaultCountdown, 'B5: CHỜ ' + remain + 's');
                if (remain === 0) {
                    log('B5: countdown het, chuyen get-code');
                    setState(STATE.GET_CODE);
                    return Promise.resolve();
                }
            }
            return Promise.resolve();
        }
        if (st === STATE.GET_CODE) {
            var attempts = parseInt(S.get('codeAttempts', '0'), 10);
            attempts++;
            S.set('codeAttempts', attempts.toString());
            var code = extractCode();
            if (code) {
                S.set('code', code);
                S.set('codeFoundAt', Date.now().toString());
                if (IS_TOP) showCode(code);
                setState(STATE.BACK_GTRAFFIC);
                setTimeout(goBackToGtraffic, CFG.backDelay);
                return Promise.resolve();
            }
            if (attempts % 10 === 0) log('Dang cho code... (lan ' + attempts + ')');
            return Promise.resolve();
        }
        if (st === STATE.BACK_GTRAFFIC) {
            if (S.get('navigatingBack') === '1') return Promise.resolve();
            goBackToGtraffic();
            return Promise.resolve();
        }
        return Promise.resolve();
    }

    function loop() {
        if (loopRunning) return;
        if (stopRequested) return;
        loopRunning = true;
        var i = 0;
        var tick = function() {
            if (stopRequested) { loopRunning = false; return; }
            if (S.get('waitingGetLink') === '1') {
                loopTimer = setTimeout(tick, 5);
                return;
            }
            if (S.get('hardStop') === '1') { loopRunning = false; return; }
            i++;
            if (i >= 999999) { loopRunning = false; return; }
            if (!shouldToolRunOnThisPage()) {
                resetAllState();
                loopRunning = false;
                return;
            }
            try { watchdog(); } catch (e) {}
            var p;
            var nextDelay = CFG.poll;
            if (isGoogle && IS_TOP) { p = handleGoogle(); nextDelay = CFG.googlePoll; }
            else if (isGtraffic && IS_TOP) { p = handleGtraffic(); nextDelay = CFG.poll; }
            else { p = handleTarget(); nextDelay = CFG.targetPoll; }
            p.catch(function(e){ log('Loi: ' + e.message); }).then(function() {
                if (getState() === STATE.DONE && IS_TOP && S.get('waitingGetLink') !== '1') { loopRunning = false; return; }
                if (S.get('hardStop') === '1' && S.get('waitingGetLink') !== '1') { loopRunning = false; return; }
                loopTimer = setTimeout(tick, nextDelay);
            });
        };
        tick();
    }

    function startGtrafficObserver() {
        if (!IS_TOP || !isGtraffic) return;
        if (gtrafficCheckTimer) return;
        gtrafficCheckTimer = setInterval(function() {
            if (S.get('waitingGetLink') === '1') {
                clearInterval(gtrafficCheckTimer); gtrafficCheckTimer = null;
                return;
            }
            if (S.get('hardStop') === '1') {
                clearInterval(gtrafficCheckTimer); gtrafficCheckTimer = null;
                return;
            }
            if (stopRequested || fillDone || S.get('confirmed') === '1') {
                clearInterval(gtrafficCheckTimer); gtrafficCheckTimer = null;
                return;
            }
            if (getState() === STATE.DONE) {
                clearInterval(gtrafficCheckTimer); gtrafficCheckTimer = null;
                return;
            }

            var code = getCodeFromPanel() || getCodeFromStorage();
            if (!code || !isValidCodeShape(code) || isBlacklistedCode(code)) return;

            var inp = findCodeInputCached();
            if (!inp) {
                return;
            }
            var curVal = (inp.value || '').trim();
            if (curVal === '' && !fillRunning && !fillDone) {
                fillAndConfirm(true);
            }
        }, 500);
    }

    function boot() {
        if (!shouldToolRunOnThisPage()) {
            var old = document.getElementById('mp-root'); if (old) old.remove();
            var fab = document.getElementById('mp-fab'); if (fab) fab.remove();
            return;
        }
        var leavingAt = parseInt(S.get('leavingAt', '0'), 10);
        if (leavingAt > 0) {
            var elapsed = Date.now() - leavingAt;
            if (elapsed > 300000) {
                resetAllState(true);
            }
            S.del('leavingAt');
        }
        if (IS_TOP) buildPanel();
        var st = getState();
        var code = S.get('code', '');
        if (isRealTargetPage() && (st === STATE.GOOGLE_SEARCH || st === STATE.GOOGLE_CLICK)) {
            setState(STATE.SCAN_BTN);
        }
        if (isGoogle && IS_TOP) {
            startGoogleObserver();
        }
        if (isGtraffic && code) {
            if (st === STATE.BACK_GTRAFFIC) {
                S.del('navigatingBack');
                S.set('backDone', '1');
            }
            if (S.get('waitingGetLink') === '1') {
                log('BOOT: dang cho nut LAY LINK tu lan truoc → tiep tuc cho');
                if (IS_TOP) refreshStatus();
                if (IS_TOP && !codeShown) {
                    codeShown = true;
                    UI.code.classList.add('show');
                    UI.codeVal.textContent = code;
                }
                startWaitingForGetLink();
                startLeaveDetector();
                return;
            }
            if (S.get('confirmed') !== '1' && S.get('hardStop') !== '1') {
                resetFillFlags();
                S.del('filled');
                if (st !== STATE.FILL_CODE) setState(STATE.FILL_CODE);
                setTimeout(function() { fillAndConfirm(true); }, 800);
                startGtrafficObserver();
            } else {
                log('BOOT: da confirmed - cho nut LAY LINK');
            }
            if (IS_TOP) refreshStatus();
            if (IS_TOP && !codeShown) {
                codeShown = true;
                UI.code.classList.add('show');
                UI.codeVal.textContent = code;
            }
            startLeaveDetector();
            if (S.get('confirmed') !== '1' && S.get('hardStop') !== '1') {
                loop();
            }
            return;
        }
        if (IS_TOP) refreshStatus();
        if (IS_TOP && code && !codeShown) {
            codeShown = true;
            UI.code.classList.add('show');
            UI.codeVal.textContent = code;
        }
        if (st === STATE.DONE && S.get('waitingGetLink') !== '1') { stopAll(); startLeaveDetector(); return; }
        if (st && st !== STATE.IDLE) loop();
        startLeaveDetector();
    }

    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', boot);
    else boot();
})();
